cmake_minimum_required(VERSION 3.24)
project(AnalogInputTests LANGUAGES CXX)
enable_testing()
add_executable(analog_input_tests main.cpp)
set(CVR_RED4EXT_SDK_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../externals/RED4ext.SDK" CACHE PATH "RED4ext SDK root")
target_include_directories(analog_input_tests PRIVATE ../../include "${CVR_RED4EXT_SDK_DIR}/include")
target_compile_features(analog_input_tests PRIVATE cxx_std_20)
foreach(case curve ranges gestures invalid default keypad keypad_cursor weapon_trigger crouch_pulse crouch_scanner_handoff snap_travel snap_boundary snap_invalid snap_hysteresis snap_native)
    add_test(NAME analog_input_${case} COMMAND analog_input_tests ${case})
endforeach()
