cmake_minimum_required(VERSION 3.24)
project(PhysicsContactTests LANGUAGES CXX)
enable_testing()
find_package(Threads REQUIRED)
add_executable(physics_contact_tests main.cpp)
target_compile_features(physics_contact_tests PRIVATE cxx_std_20)
target_include_directories(physics_contact_tests PRIVATE ../../include)
target_link_libraries(physics_contact_tests PRIVATE Threads::Threads)
# Assertions are the checks, including Release configurations.
if(MSVC)
    target_compile_options(physics_contact_tests PRIVATE /UNDEBUG)
else()
    target_compile_options(physics_contact_tests PRIVATE -UNDEBUG)
endif()
add_test(NAME physics_contact COMMAND physics_contact_tests)
if(WIN32)
    add_executable(native_geometry_tests native_geometry.cpp)
    target_compile_features(native_geometry_tests PRIVATE cxx_std_20)
    target_include_directories(native_geometry_tests PRIVATE ../../include)
    target_compile_definitions(native_geometry_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
    target_compile_options(native_geometry_tests PRIVATE /UNDEBUG)
endif()
